> **给执行者:** 按此计划逐任务实现,按步骤顺序执行,步骤用复选框(`- [ ]`)语法跟踪进度。
> 本计划**不含仓库内测试文件**(仓库纯净原则);验证 = 既有行为套件 + 仓库外临时脚本(d:/tmp,用完即删) + 手动 curl e2e。
> 本计划已经 plan-verifier 执行前审查修订(2026-09-10):代码锚点/签名/环境依赖逐项核实通过;行为套件路径修正为 `teage_liu2\PROTOCOL\behavior-suite\runner.py`;curl 统一 `curl.exe`;补端口定向停服命令。

**目标:** 把 core 对话读路径的同步磁盘 IO 移出事件循环(P0),并把 stdio 存储代理的每轮读 IPC 从 2 次降到 1 次(P1 短路),同时建立 per-op 耗时指标,为 P2(批量写/流水线)立项提供基线数据。

**架构:** ①`pipeline.chat_stream` 的 `ensure_session + get_session_messages` 合并为单次 `asyncio.to_thread` 调用(不改 HistoryStore 契约,默认 SQLite 与 stdio 后端同时受益);②`StdioStorageProxy.ensure_session` 利用幂等语义(INSERT OR IGNORE)做进程内短路;③代理内建 per-op 计时聚合,经 `/health` 暴露。

**技术栈:** Python 3.11(asyncio/ threading)/ FastAPI / 既有 P-4 storage-stdio 协议(JSON Lines,Rust 后端零改动)。

### 任务 2: StdioStorageProxy ensure_session 幂等短路

**文件:**
- 修改: `teage_liu2/server/storage_stdio_proxy.py`(`StdioStorageProxy.__init__` + `ensure_session`)

**接口:**
- 消费: 既有 `self._call(op, payload)`;任务 3 将给 `_call` 加计时(签名不变)
- 产出: 短路语义 —— 同一 session_id 首次成功后,后续 `ensure_session` 零 IPC 直接返回

**语义说明(写进代码注释):**
- 仅在请求成功后标记;失败(超时/断连)不标记,下次重试;
- 后端连接生命周期 = 代理实例生命周期,断连后所有 `_call` 抛 RuntimeError、连接原地不重建,故集合不会失效;
- 集合增长上界 = 会话数(每会话一个短字符串),不设淘汰。

- [ ] **步骤 1: `__init__` 增加短路状态**

`StdioStorageProxy.__init__` 末尾(`self._close_lock = threading.Lock()` 之后)追加:

```python
        # 幂等短路:ensure_session 后端语义为 INSERT OR IGNORE(幂等),
        # 同会话首次成功后进程内标记,后续调用零 IPC(P-1)。
        # 仅成功后标记;连接断开 = 代理整体失效(不原地重建),集合不会失效。
        self._ensured_sessions: set = set()
        self._ensured_lock = threading.Lock()
```

- [ ] **步骤 2: 重写 `ensure_session` 方法**

原实现(第 326-327 行):

```python

        self._call("ensure_session", {"session_id": session_id})
```

替换为:

```python
    def ensure_session(self, session_id: str) -> None:
        with self._ensured_lock:
            if session_id in self._ensured_sessions:
                return
        self._call("ensure_session", {"session_id": session_id})
        with self._ensured_lock:
            self._ensured_sessions.add(session_id)
```

```python
class _OpMetrics:
    """per-op 耗时统计(线程安全轻量聚合,诊断/健康检查用,不改变行为)。

    统计口径 = 宿主调用线程视角的端到端耗时(含线程投递 + IPC 往返)。
    """

    __slots__ = ("_lock", "_data")

    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._data: Dict[str, Dict[str, Any]] = {}

    def record(self, op: str, duration_ms: float, ok: bool) -> None:
        with self._lock:
            d = self._data.setdefault(
                op, {"count": 0, "total_ms": 0.0, "max_ms": 0.0, "errors": 0}
            )
            d["count"] += 1
            d["total_ms"] += duration_ms
            if duration_ms > d["max_ms"]:
                d["max_ms"] = duration_ms
            if not ok:
                d["errors"] += 1

    def snapshot(self) -> Dict[str, Dict[str, Any]]:
        with self._lock:
            return {
                op: {**d, "avg_ms": round(d["total_ms"] / d["count"], 3)}
                for op, d in self._data.items()
            }
```

```python
"""存储通道基线测量(仓库外临时件,用完即删)。
对比:进程内 SQLiteHistoryStore vs StdioStorageProxy(storage_rust)。
"""
import os
import shutil
import statistics
import sys
import time

REPO = r"d:\dev\teage-liu\teage-liu"
sys.path.insert(0, REPO)

from teage_liu2.core.history import SQLiteHistoryStore
from teage_liu2.server.storage_stdio_proxy import StdioStorageProxy

N = 200
TMP = r"D:\tmp\bench_data"
EXE = os.path.join(REPO, r"data2\extensions\storage_rust\teage-storage-rust.exe")


def bench(name: str, store) -> dict:
    sid = f"bench_{name}"
    store.ensure_session(sid)
    out = {}
    for op, fn in (
        ("log_message", lambda i: store.log_message(sid, "user", f"msg-{i}", token_count=5)),
        ("get_session_messages", lambda i: store.get_session_messages(sid, limit=50)),
    ):
        ts = []
        for i in range(N):
            t0 = time.perf_counter()
            fn(i)
            ts.append((time.perf_counter() - t0) * 1000.0)
        ts.sort()
        out[op] = {
            "mean": round(statistics.mean(ts), 3),
            "p50": round(ts[N // 2], 3),
            "p95": round(ts[int(N * 0.95)], 3),
            "max": round(ts[-1], 3),
        }
    snap = getattr(store, "metrics_snapshot", None)
    if callable(snap):
        out["_proxy_metrics"] = snap()
    return out


def main() -> None:
    os.makedirs(TMP, exist_ok=True)
    try:
        sqlite_db = os.path.join(TMP, "bench_sqlite.db")
        rust_db = os.path.join(TMP, "bench_rust.db")
        results = {}
        s = SQLiteHistoryStore(sqlite_db)
        try:
            results["sqlite_inprocess"] = bench("sqlite", s)
        finally:
            s.close()
        proxy = StdioStorageProxy([EXE, "--db", rust_db], request_timeout=10.0)
        try:
            proxy.start()
            results["stdio_proxy_rust"] = bench("rust", proxy)
        finally:
            proxy.close()
        for backend, ops in results.items():
            print(f"\n== {backend} ==")
            for op, st in ops.items():
                print(f"  {op}: {st}")
    finally:
        shutil.rmtree(TMP, ignore_errors=True)


if __name__ == "__main__":
    main()
```
