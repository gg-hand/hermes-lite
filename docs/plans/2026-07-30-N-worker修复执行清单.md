```python
# file_lock.py 新增
import portalocker
import asyncio
from contextlib import asynccontextmanager
from pathlib import Path

class FileLock:
    """跨进程文件锁，基于 portalocker.LOCK_EX"""
    def __init__(self, path: Path, timeout: float = 5.0):
        self._path = path
        self._timeout = timeout
        self._lock = None

    async def __aenter__(self):
        # portalocker.Lock 是同步的，丢到 executor
        loop = asyncio.get_event_loop()
        def _acquire():
            self._lock = portalocker.Lock(str(self._path), timeout=self._timeout)
            self._lock.acquire()
        await loop.run_in_executor(None, _acquire)
        return self

    async def __aexit__(self, *args):
        if self._lock:
            self._lock.release()
```

### 任务 1.3 director_engine._run_loop 异常容错（类别 B1）

**文件**：`teage_liu/multiagent/director_engine.py` (L311-332)

**改动**：每个子任务独立 try/except，失败计数；连续失败 10 次触发重启

**代码骨架**：
```python
async def _run_loop(self):
    tasks = [
        ("tick",          self._update_director_tick),
        ("heartbeat",     self._check_worker_heartbeats),
        ("turn_timeout",  self._check_turn_timeout),
        ("observe",       self._observe_and_intervene),
        ("flush_pending", self._flush_pending_messages),
        ("arbitrate",     self._arbitrate_conflicts),
    ]
    self._task_failure_counts = {name: 0 for name, _ in tasks}
    while self._running:
        try:
            await asyncio.sleep(self._tick_interval)
            for name, fn in tasks:
                try:
                    await fn()
                except asyncio.CancelledError:
                    raise
                except Exception as e:
                    logger.warning("Director 子任务 %s 失败: %s", name, e, exc_info=True)
                    self._task_failure_counts[name] += 1
                    if self._task_failure_counts[name] > 10:
                        logger.error("子任务 %s 连续失败 10 次，触发 director 重启", name)
                        self._running = False
                        return
            # 成功一轮，衰减失败计数
            for name in self._task_failure_counts:
                if self._task_failure_counts[name] > 0:
                    self._task_failure_counts[name] -= 1
        except asyncio.CancelledError:
            raise
        except Exception as e:
            logger.exception("Director 主循环异常: %s", e)
```

### 任务 1.5 LocalDirectorManager 自动重启（类别 B3）

**文件**：`teage_liu/multiagent/director_manager.py`

**改动**：增加 `_watchdog` 协程，每 10s 检查子进程，崩溃自动重启

**代码骨架**：
```python
class LocalDirectorManager:
    def __init__(self, ...):
        # ... 已有字段
        self._watchdog_task = None
        self._watchdog_running = False
        self._restart_count = 0
        self._max_restarts = config.get("max_restarts", 3)
        self._restart_window = config.get("restart_window_seconds", 300)
        self._restart_times = []  # 最近重启时间列表

    async def start(self):
        # ... 已有启动逻辑
        self._watchdog_running = True
        self._watchdog_task = asyncio.create_task(self._watchdog())

    async def _watchdog(self):
        while self._watchdog_running:
            await asyncio.sleep(10)
            if self._process is None:
                continue
            rc = self._process.poll()
            if rc is not None:
                logger.error("Director 子进程退出 rc=%s", rc)
                self._cleanup_stale_state()
                # 检查重启窗口
                now = time.time()
                self._restart_times = [t for t in self._restart_times if now - t < self._restart_window]
                if len(self._restart_times) < self._max_restarts:
                    try:
                        await asyncio.sleep(5)  # 重启前等待
                        self.start()
                        self._restart_times.append(now)
                        logger.info("Director 自动重启成功 (尝试 %s/%s)", len(self._restart_times), self._max_restarts)
                    except Exception as e:
                        logger.exception("自动重启失败: %s", e)
                else:
                    logger.error("已达最大重启次数 %s，停止 watchdog", self._max_restarts)
                    self._watchdog_running = False
                    break

    def _cleanup_stale_state(self):
        """清理上次崩溃留下的锁文件"""
        for lock_name in ["director.lock", "audit.lock"]:
            lock_path = os.path.join(self._bb_root, "locks", lock_name)
            if os.path.exists(lock_path):
                try:
                    os.remove(lock_path)
                except PermissionError:
                    logger.warning("无法删除 %s，可能仍被持有", lock_path)
        if os.path.exists(self._pid_file):
            try:
                os.remove(self._pid_file)
            except Exception:
                pass
```

## 风险与回滚

### 风险点

| 风险 | 概率 | 缓解 |
|---|---|---|
| 加锁引入死锁 | 中 | 锁超时 5s + task_failure_counts 触发重启 |
| 选举算法误判 | 中 | epoch + 字典序 + lease 30s；启动时强制 Election.run() |
| 自动重启无限循环 | 低 | _max_restarts=3 / 5 分钟窗口 |
| watchdog 在多机场景失效 | 中 | 已有自检 + 降级轮询 |

### 回滚预案

1. **代码级**：所有改动用 `if config.get("director_v2_enabled", True):` 包住
2. **配置级**：保留 `config-a2a-X.yaml.bak`
3. **进程级**：`auto_restart_enabled` 配置开关
4. **架构级**：保留 `LocalDirectorManager` 作为 fallback
