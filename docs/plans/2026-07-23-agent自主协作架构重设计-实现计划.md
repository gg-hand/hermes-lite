# Agent 自主协作架构重设计 实现计划

> **For agentic workers:** 使用 TDD 工作流逐任务实现此计划。
> 步骤使用复选框（`- [ ]`）语法跟踪进度。

**Goal:** 将 Director 中心化任务分派模式重构为 Agent 自主协作 + Director 按需插入的去中心化模式

**Architecture:** 物理分离 messages.md（主对话）和 collaboration.md（协作空间）+ collabs/{collab_id}.md（单协作）；Agent 通过 A2A + 黑板混合通信；Director 接口化，默认静默观察，按需插入 directive；工作台作为透明观察窗口

**Tech Stack:** Python 3.11+ / FastAPI / asyncio / pytest / 原生 JS + CSS

## Global Constraints

- Python 3.11+，asyncio 异步
- 测试用 pytest，遵循 TDD（红→绿→重构→提交）
- 前端原生 JS，不引入框架
- CSS 变量复用 tokens.css 已有定义
- 后端串行化写入（asyncio.Lock），不依赖文件锁
- 不向后兼容旧 task/assign/status/result 消息类型
- 文件路径用 pathlib.Path，禁止硬编码绝对路径
- API 密钥不硬编码

### 修复 D5：Task 12 CollabWriter 死锁风险

**问题**：Task 12 描述"在 CollabWriter.append 成功写入后添加索引更新"，但 append 在锁内，调用 update_collab_index 又会尝试获取同一个锁 → 死锁。同时 update_collab_index 用 `async with writer._lock` 访问私有属性。

**修复**：
1. 索引更新移到锁外：CollabWriter.append 返回后，由调用方（append_collab_message 包装函数）执行索引更新
2. update_collab_index 不再获取 writer._lock，直接用 atomic_write 原子写入（atomic_write 本身用唯一 .tmp 文件，配合 append_collab_message 的串行化保证一致性）

```python
# append_collab_message 包装函数
async def append_collab_message(bb_root, message, collab_id=None) -> tuple[int, bool]:
    writer = _get_global_writer(bb_root)
    seq, deduplicated = await writer.append(message, collab_id)
    # 锁外执行索引更新（仅当写入成功且非去重时）
    if not deduplicated and collab_id and message.get("type") == "request":
        try:
            await update_collab_index(bb_root, collab_id,
                title=message.get("content", "")[:50],
                status="initiated",
                participants=[message.get("from", "")])
        except Exception:
            pass  # 索引更新失败不影响消息写入
    return seq, deduplicated


# update_collab_index 不再获取 writer._lock
async def update_collab_index(bb_root, collab_id, title, status, participants) -> None:
    """更新协作索引（upsert，原子写入）"""
    index_path = bb_root / "collabs" / "index.md"
    index_path.parent.mkdir(parents=True, exist_ok=True)
    entries = await read_collab_index(bb_root)
    # upsert
    found = False
    for i, e in enumerate(entries):
        if e.get("collab_id") == collab_id:
            entries[i] = {"collab_id": collab_id, "title": title,
                          "status": status, "participants": participants}
            found = True
            break
    if not found:
        entries.append({"collab_id": collab_id, "title": title,
                        "status": status, "participants": participants})
    # 原子写入（atomic_write 用唯一 .tmp，配合串行化保证一致性）
    content = ""
    for e in entries:
        content += f"---\n{yaml.safe_dump(e, allow_unicode=True, default_flow_style=False, sort_keys=False)}---\n\n"
    await atomic_write(index_path, content)
```

### 修复 A1：Task 2 复用现有 frontmatter 解析逻辑

**问题**：Task 2 的 `read_collab_messages`/`_read_last_collab_seq` 重新实现了 `read_messages`/`_read_last_message_seq` 的解析逻辑，违反 DRY。

**修复**：在 `blackboard.py` 中先抽取 `_parse_frontmatter_blocks(content: str) -> list[dict]` 辅助函数，让 `read_messages` 和 `read_collab_messages` 共用。

```python
# blackboard.py 新增辅助函数（在 read_messages 之前）
def _parse_frontmatter_blocks(content: str) -> list[dict]:
    """解析多个 YAML frontmatter 块（每个以 --- 分隔）。

    供 read_messages 和 read_collab_messages 共用，避免逻辑重复。
    """
    if not content.strip():
        return []
    records: list[dict] = []
    parts = content.split("---\n")
    for i in range(1, len(parts), 2):
        if i >= len(parts):
            break
        frontmatter_str = parts[i]
        if not frontmatter_str.strip():
            continue
        try:
            frontmatter = yaml.safe_load(frontmatter_str)
            if isinstance(frontmatter, dict):
                records.append(frontmatter)
        except yaml.YAMLError:
            continue
    return records


# 同时重构 read_messages 复用上述函数
async def read_messages(bb_root: Path) -> list[dict]:
    messages_path = bb_root / "messages.md"
    if not messages_path.exists():
        return []
    return _parse_frontmatter_blocks(messages_path.read_text(encoding="utf-8"))


# collab 版本
async def read_collab_messages(bb_root: Path, collab_id: Optional[str] = None) -> list[dict]:
    file_path = _get_collab_file(bb_root, collab_id)
    if not file_path.exists():
        return []
    return _parse_frontmatter_blocks(file_path.read_text(encoding="utf-8"))


async def _read_last_collab_seq(bb_root: Path, collab_id: Optional[str] = None) -> int:
    messages = await read_collab_messages(bb_root, collab_id)
    if not messages:
        return 0
    return max(m.get("seq", 0) for m in messages if isinstance(m.get("seq"), int))
```

### 修复 A2：Task 3 /forward 锁不互斥 + CollabWriter 返回值

**问题**：`/forward` 端点 `CollabWriter(bb_root)` 创建新实例，与全局 `_get_global_writer` 锁不互斥；调用私有方法 `_find_by_message_id`；双重查询效率低。

**修复**：`CollabWriter.append` 返回 `(seq, deduplicated: bool)` 元组，`/forward` 使用全局 writer。

```python
# CollabWriter.append 改为返回元组
class CollabWriter:
    async def append(self, message: dict, collab_id: Optional[str] = None) -> tuple[int, bool]:
        """串行化写入，返回 (seq, deduplicated)"""
        async with self._lock:
            message_id = message.get("message_id")
            if message_id:
                existing = await self._find_by_message_id(message_id, collab_id)
                if existing is not None:
                    return existing["seq"], True  # 去重
            last_seq = await _read_last_collab_seq(self._bb_root, collab_id)
            message = {**message, "seq": last_seq + 1}
            file_path = _get_collab_file(self._bb_root, collab_id)
            import yaml
            block = f"---\n{yaml.dump(message, allow_unicode=True, default_flow_style=False)}\n"
            with open(file_path, "a", encoding="utf-8") as f:
                f.write(block)
            return message["seq"], False


# append_collab_message 包装函数也更新返回值
async def append_collab_message(bb_root: Path, message: dict, collab_id: Optional[str] = None) -> tuple[int, bool]:
    writer = _get_global_writer(bb_root)
    return await writer.append(message, collab_id)

@router.post("/forward")

        "from": req.from_,

# ========== 协作消息（collaboration.md + collabs/{collab_id}.md） ==========

import asyncio
from pathlib import Path
from typing import Optional


def _get_collab_file(bb_root: Path, collab_id: Optional[str] = None) -> Path:
    """获取协作消息文件路径"""
    if collab_id:
        collabs_dir = bb_root / "collabs"
        collabs_dir.mkdir(parents=True, exist_ok=True)
        return collabs_dir / f"{collab_id}.md"
    else:
        return bb_root / "collaboration.md"


# 消息文件格式：每条消息用 --- 分隔，格式如下：
# ---
# from: agent_A
# type: announce
# content: 我上线了
# seq: 1
# ---
# from: agent_B
# type: relay
# content: 你好
# seq: 2
async def _read_last_collab_seq(bb_root: Path, collab_id: Optional[str] = None) -> int:
    """读取最后一条协作消息的 seq"""
    file_path = _get_collab_file(bb_root, collab_id)
    if not file_path.exists():
        return 0
    messages = await read_collab_messages(bb_root, collab_id)
    if not messages:
        return 0
    return max(m.get("seq", 0) for m in messages)


async def read_collab_messages(
    bb_root: Path, collab_id: Optional[str] = None
) -> list[dict]:
    """读取协作消息"""
    file_path = _get_collab_file(bb_root, collab_id)
    if not file_path.exists():
        return []
    # 复用现有的消息解析逻辑
    content = file_path.read_text(encoding="utf-8")
    messages = []
    # 按 --- 分割消息块，每个块是一个 YAML 文档
    blocks = content.split("---\n")
    for block in blocks:
        block = block.strip()
        if not block:
            continue
        try:
            import yaml
            msg = yaml.safe_load(block)
            if msg and isinstance(msg, dict):
                messages.append(msg)
        except Exception:
            continue
    messages.sort(key=lambda m: m.get("seq", 0))
    return messages


async def append_collab_message(
    bb_root: Path, message: dict, collab_id: Optional[str] = None
) -> int:
    """写入协作消息（串行化，带 message_id 去重）

    Returns:
        分配的 seq。如果 message_id 已存在，返回已存在的 seq。
    """
    writer = _get_global_writer(bb_root)
    return await writer.append(message, collab_id)


class CollabWriter:
    """协作消息写入器，使用 asyncio.Lock 串行化所有写入"""

    def __init__(self, bb_root: Path):
        self._bb_root = bb_root
        self._lock = asyncio.Lock()

    async def append(
        self, message: dict, collab_id: Optional[str] = None
    ) -> int:
        """串行化写入消息"""
        async with self._lock:
            # 检查 message_id 去重
            message_id = message.get("message_id")
            if message_id:
                existing = await self._find_by_message_id(message_id, collab_id)
                if existing is not None:
                    return existing["seq"]

            # 分配 seq
            last_seq = await _read_last_collab_seq(self._bb_root, collab_id)
            message = {**message, "seq": last_seq + 1}

            # 写入文件
            file_path = _get_collab_file(self._bb_root, collab_id)
            import yaml
            block = f"---\n{yaml.dump(message, allow_unicode=True, default_flow_style=False)}\n"
            with open(file_path, "a", encoding="utf-8") as f:
                f.write(block)

            return message["seq"]

    async def _find_by_message_id(
        self, message_id: str, collab_id: Optional[str] = None
    ) -> Optional[dict]:
        """查找已存在的相同 message_id 消息"""
        messages = await read_collab_messages(self._bb_root, collab_id)
        for m in messages:
            if m.get("message_id") == message_id:
                return m
        return None


# 全局 CollabWriter 单例（按 bb_root 缓存）
_writers: dict[Path, CollabWriter] = {}


def _get_global_writer(bb_root: Path) -> CollabWriter:
    """获取全局 CollabWriter 单例"""
    if bb_root not in _writers:
        _writers[bb_root] = CollabWriter(bb_root)
    return _writers[bb_root]

"""协作消息 API 路由"""
from pathlib import Path
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import Optional
from teage_liu.multiagent.blackboard import (
    read_collab_messages,
    append_collab_message,
)

router = APIRouter(prefix="/api/multiagent/collab", tags=["collaboration"])


def _get_bb_root() -> Path:
    """获取黑板根目录"""
    import os
    return Path(os.environ.get("TEAGE_BB_ROOT", "data/blackboard"))


class AppendRequest(BaseModel):
    message: dict
    collab_id: Optional[str] = None


class ForwardRequest(BaseModel):
    message_id: str
    from_: str
    to: str
    content: str
    via: str = "a2a"


class BroadcastRequest(BaseModel):
    content: str


class AnnounceRequest(BaseModel):
    agent_id: str
    action: str  # online/offline/capability_update
    capabilities: list[str] = []
    endpoint: str = ""
    agent_name: str = ""


@router.get("/messages")
async def get_messages(collab_id: Optional[str] = None):
    """获取协作消息列表"""
    bb_root = _get_bb_root()
    messages = await read_collab_messages(bb_root, collab_id)
    return {"messages": messages, "total": len(messages)}


@router.post("/append")
async def append_message(req: AppendRequest):
    """写入协作消息"""
    bb_root = _get_bb_root()
    seq = await append_collab_message(bb_root, req.message, req.collab_id)
    return {"ok": True, "seq": seq}


@router.post("/forward")
async def forward_message(req: ForwardRequest):
    """转发 A2A 消息到协作黑板"""
    bb_root = _get_bb_root()
    # 构造转发消息
    message = {
        "from": req.from_,
        "to": req.to,
        "type": "relay",
        "content": req.content,
        "via": req.via,
        "message_id": req.message_id,
    }
    # 检查是否已存在
    from teage_liu.multiagent.blackboard import CollabWriter
    writer = CollabWriter(bb_root)
    existing = await writer._find_by_message_id(req.message_id, None)
    if existing is not None:
        return {"ok": True, "seq": existing["seq"], "deduplicated": True}

    seq = await append_collab_message(bb_root, message)
    return {"ok": True, "seq": seq, "deduplicated": False}


@router.post("/broadcast")
async def broadcast(req: BroadcastRequest):
    """用户发布广播（from=user 的 request）"""
    bb_root = _get_bb_root()
    message = {
        "from": "user",
        "type": "request",
        "content": req.content,
        "to": "*",
    }
    seq = await append_collab_message(bb_root, message)
    return {"ok": True, "seq": seq}


@router.post("/announce")
async def announce(req: AnnounceRequest):
    """agent 上线/下线/能力更新"""
    bb_root = _get_bb_root()
    message = {
        "from": req.agent_id,
        "type": "announce",
        "action": req.action,
        "capabilities": req.capabilities,
        "endpoint": req.endpoint,
        "agent_name": req.agent_name,
    }
    seq = await append_collab_message(bb_root, message)
    return {"ok": True, "seq": seq}


@router.get("/agents")
async def get_agents():
    """获取接入工作台的 agent 列表（从 announce 消息提取）"""
    bb_root = _get_bb_root()
    messages = await read_collab_messages(bb_root)
    # 从 announce 消息提取 agent 列表
    agents = {}
    for msg in messages:
        if msg.get("type") == "announce":
            agent_id = msg.get("from", "")
            action = msg.get("action", "")
            if action == "online":
                agents[agent_id] = {
                    "agent_id": agent_id,
                    "agent_name": msg.get("agent_name", agent_id),
                    "capabilities": msg.get("capabilities", []),
                    "endpoint": msg.get("endpoint", ""),
                    "status": "active",
                }
            elif action == "offline" and agent_id in agents:
                agents[agent_id]["status"] = "inactive"
    return {"agents": list(agents.values())}

from fastapi.responses import StreamingResponse
import asyncio
import json
import time


@router.get("/sse")
async def collab_sse():
    """协作消息 SSE 通道"""
    bb_root = _get_bb_root()

    async def event_stream():
        last_seq = 0
        # 初始发送已有消息
        messages = await read_collab_messages(bb_root)
        for msg in messages:
            yield f"data: {json.dumps({'type': 'collab_message_append', 'data': {'message': msg}}, ensure_ascii=False)}\n\n"
            last_seq = max(last_seq, msg.get("seq", 0))

        # 轮询新消息
        while True:
            await asyncio.sleep(1)
            messages = await read_collab_messages(bb_root)
            new_messages = [m for m in messages if m.get("seq", 0) > last_seq]
            for msg in new_messages:
                yield f"data: {json.dumps({'type': 'collab_message_append', 'data': {'message': msg}}, ensure_ascii=False)}\n\n"
                last_seq = msg["seq"]

    return StreamingResponse(
        event_stream(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )

"""Director 接口协议"""
from abc import ABC, abstractmethod
from pathlib import Path


class DirectorProtocol(ABC):
    """Director 协议：定义干预 agent 协作的标准操作。

    Director 是"按需插入者"，默认静默观察，只在需要时插入 directive。
    三种实现：AgentDirector（LLM驱动）/ UserDirector（用户手动）/ ScriptDirector（脚本规则）
    """

    @abstractmethod
    async def inject_directive(
        self, content: str, rule_type: str, target: str = "*"
    ) -> None:
        """主动注入协作引导（双通道）。

        1. 写入 collaboration.md（可审计记录）
        2. 注入目标 agent 的 LLM 对话上下文（agent 自然感知）

        Args:
            content: 引导内容（自然语言描述的协作约束）
            rule_type: ordering（顺序约束）/ constraint（规则约束）/ intervention（应急干预）
            target: 目标 agent_id 或 "*"（全部）
        """

    @abstractmethod
    async def handle_arbitration(self, request: dict) -> dict:
        """处理 agent 的仲裁请求（被动响应）。

        当 agent 间出现分歧时，可向 Director 发起仲裁请求。
        """

    @abstractmethod
    async def observe(self) -> dict:
        """观察协作状态（用于异常检测）。"""
```

        anomalies = []
        messages = await read_collab_messages(self._bb_root)
        now = time.time()

        # 检测即时型 request 超时
        for msg in messages:
            if (
                msg.get("type") == "request"
                and msg.get("collab_type") == "instant"
                and msg.get("timeout")
            ):
                # 查找是否有对应的 result
                has_result = any(
                    m.get("type") == "result"
                    and m.get("reply_to") == msg.get("seq")
                    for m in messages
                )
                if not has_result:
                    # 简化：用 seq 推断写入时间（实际应记录时间戳）
                    # 这里用 timeout 字段判断
                    anomalies.append({
                        "type": "timeout",
                        "message": msg,
                        "detail": f"request seq={msg.get('seq')} 超时未响应",

```python
# 在 WorkerAdapter 类中添加

import time
from datetime import datetime, timezone


def __init__(self, agent_id: str, bb_root: Path, orchestrator=None,
             capabilities: list = None, endpoint: str = "",
             idle_timeout_seconds: int = 60):
    # 保留现有初始化逻辑
    self._agent_id = agent_id
    self._bb_root = bb_root
    self._orchestrator = orchestrator
    self._capabilities = capabilities or []
    self._endpoint = endpoint
    self._collab_poll_task = None
    self._idle_check_task = None

    # 双队列（A2A 主泵方案 §12.2）
    self._normal_queue = []      # 普通队列：directive(ordering/constraint)、relay、request(非user)
    self._urgent_queue = []      # 紧急队列：directive(intervention)、用户广播

    self._last_collab_seq = 0    # 已处理的协作消息 seq
    self._last_a2a_time = time.time()  # 最后一次 A2A 调用时间
    self._idle_timeout_seconds = idle_timeout_seconds  # 空闲超时阈值


async def _write_announce(self, action: str) -> None:
    """写 announce 消息（上线/下线）"""
    from teage_liu.multiagent.blackboard import append_collab_message
    message = {
        "from": self._agent_id,
        "type": "announce",
        "action": action,
        "capabilities": self._capabilities,
        "endpoint": self._endpoint,
    }
    await append_collab_message(self._bb_root, message)


async def _collab_poll_loop(self) -> None:
    """协作消息轮询主循环"""
    while True:
        try:
            await self._poll_collab_once()
        except Exception as e:
            # 日志记录
            pass
        await asyncio.sleep(2)  # 2秒轮询间隔


async def _poll_collab_once(self) -> None:
    """轮询一次协作消息"""
    from teage_liu.multiagent.blackboard import read_collab_messages
    messages = await read_collab_messages(self._bb_root)
    new_messages = [m for m in messages if m.get("seq", 0) > self._last_collab_seq]

    for msg in new_messages:
        await self._handle_collab_message(msg)
        self._last_collab_seq = max(self._last_collab_seq, msg.get("seq", 0))


async def _handle_collab_message(self, msg: dict) -> None:
    """处理单条协作消息（按消息分类路由）"""
    msg_type = msg.get("type")

    if msg_type == "directive":
        await self._handle_directive(msg)
    elif msg_type == "request":
        await self._handle_request(msg)
    elif msg_type == "announce":
        await self._handle_announce(msg)
    elif msg_type == "relay":
        # relay 消息已在 A2A 中处理，只记录上下文（普通队列）
        self._normal_queue.append(msg)
    elif msg_type == "status":
        # 状态更新，不触发 LLM，不入队
        pass
    elif msg_type == "response":
        # response 进入普通队列，搭便车
        self._normal_queue.append(msg)
    elif msg_type == "result":
        # result 进入普通队列，搭便车
        self._normal_queue.append(msg)


async def _handle_request(self, msg: dict) -> None:
    """处理协作请求"""
    # 检查是否需要自己的能力
    capabilities_needed = msg.get("capabilities_needed", [])
    if capabilities_needed and not any(
        cap in self._capabilities for cap in capabilities_needed
    ):
        return  # 不匹配，跳过

    # 检查 target
    target = msg.get("target_agents", [])
    if target and self._agent_id not in target and "*" not in target:
        return  # 不是发给自己的

    # 判断是否用户广播（紧急插队）
    if msg.get("from") == "user":
        # 紧急：立即触发 LLM
        await self._trigger_urgent_llm(
            prompt=f"用户广播协作请求：{msg.get('content', '')}\n请决定是否参与并回复。",
            context_msg=msg,
        )
    else:
        # 普通：进入队列，等 A2A 主泵触发
        self._normal_queue.append(msg)


async def _handle_directive(self, msg: dict) -> None:
    """处理 Director directive（按 rule_type 分类）"""
    target = msg.get("target", "*")
    if target != "*" and target != self._agent_id:
        return  # 不是给自己的

    rule_type = msg.get("rule_type", "")
    if rule_type == "intervention":
        # 紧急：立即触发 LLM（紧急插队 §12.5）
        await self._trigger_urgent_llm(
            prompt=f"Director 应急干预：{msg.get('content', '')}\n请立即响应。",
            context_msg=msg,
        )
    else:
        # 普通：ordering/constraint 进入普通队列，搭便车
        self._normal_queue.append(msg)


async def _handle_announce(self, msg: dict) -> None:
    """处理其他 agent 的 announce（更新本地状态，不调用 LLM）"""

async def _trigger_urgent_llm(self, prompt: str, context_msg: dict) -> None:
    """紧急插队：立即触发 LLM 调用（不搭便车）"""
    # 紧急消息置于 system prompt 顶部
    system_prompt = self._build_urgent_system_prompt(context_msg)
    # 同时把普通队列消息也注入（顺便清空）
    system_prompt = self._inject_normal_queue_to_context(system_prompt)

    response = await self._orchestrator.chat(prompt, system_prompt=system_prompt)
    self._last_a2a_time = time.time()  # 更新最后调用时间

    # 如果是 request，写 response

        from teage_liu.multiagent.blackboard import append_collab_message
        await append_collab_message(self._bb_root, {
            "from": self._agent_id,
            "type": "response",
            "reply_to": context_msg.get("seq"),

            "accept": True,

def _build_urgent_system_prompt(self, context_msg: dict) -> str:
    """构造紧急消息的 system prompt（顶部置入）"""
    msg_type = context_msg.get("type")
    content = context_msg.get("content", "")
    issued_by = context_msg.get("issued_by", "")

    if msg_type == "directive" and context_msg.get("rule_type") == "intervention":
        return (
            f"[⚠️ 紧急 - Director 应急干预]\n"
            f"来源：{issued_by or 'Director'}\n"
            f"内容：{content}\n"
            f"请立即响应此干预。\n\n"
            f"你是协作 agent。"
        )
    elif context_msg.get("from") == "user":
        return (
            f"[📢 用户广播协作请求]\n"
            f"内容：{content}\n"
            f"请判断是否参与并回复。\n\n"
            f"你是协作 agent。"
        )
    return "你是协作 agent。"


async def _call_llm_with_pump(self, prompt: str, system_prompt: str) -> str:
    """A2A 主泵：LLM 调用 + 普通队列注入（搭便车）"""
    # 把普通队列消息搭便车注入
    system_prompt = self._inject_normal_queue_to_context(system_prompt)

    response = await self._orchestrator.chat(prompt, system_prompt=system_prompt)
    self._last_a2a_time = time.time()  # 更新最后调用时间
    return response


def _inject_normal_queue_to_context(self, system_prompt: str) -> str:
    """把普通队列消息注入 LLM 上下文（搭便车机制，字段级强化）"""
    if not self._normal_queue:
        return system_prompt

    # 按消息类型分组构造注入内容
    directive_parts = []
    other_parts = []

    for msg in self._normal_queue:
        if msg.get("type") == "directive":
            # directive 字段级强化（§12.7）
            priority = msg.get("priority", "normal")
            issued_by = msg.get("issued_by", "Director")
            deadline = msg.get("deadline")
            rule_type = msg.get("rule_type", "")
            content = msg.get("content", "")

            priority_label = "高优先级" if priority == "high" else "普通"
            deadline_str = f" | 期望生效：{deadline}秒内" if deadline else ""
            directive_parts.append(
                f"- [{priority_label}] {content}（类型：{rule_type}，来源：{issued_by}{deadline_str}）"
            )
        else:
            # 其他普通消息
            from_name = msg.get("from", "")
            msg_type = msg.get("type", "")
            content = msg.get("content", "")
            other_parts.append(f"- [{from_name} → {msg_type}] {content}")

    injection = ""
    if directive_parts:
        injection += "\n\n[⚠️ Director 引导]\n当前协作中存在 Director 角色，其职责是协调和引导协作。请适当遵循以下引导：\n"
        injection += "\n".join(directive_parts)
    if other_parts:
        if injection:
            injection += "\n\n"
        injection += "[📋 协作上下文]\n"
        injection += "\n".join(other_parts)

    # 清空队列
    self._normal_queue.clear()

    return system_prompt + injection

async def _check_idle_timeout(self) -> None:
    """空闲超时保护：60s 无 A2A 且队列非空 → 触发 LLM 调用"""
    if not self._normal_queue:
        return

    idle_seconds = time.time() - self._last_a2a_time
    if idle_seconds < self._idle_timeout_seconds:
        return

    # 触发 LLM 调用处理队列
    await self._call_llm_with_pump(
        prompt="[空闲超时] 队列中有待处理消息，请处理。",
        system_prompt="你是协作 agent。",
    )


async def _idle_check_loop(self) -> None:
    """空闲超时检查循环"""
    while True:
        try:
            await self._check_idle_timeout()
        except Exception:
            pass
        await asyncio.sleep(10)  # 10秒检查一次

# 在 DirectorEngine 类中

class DirectorEngine:
    """Director 引擎：观察协作状态，按需插入 directive"""

    def __init__(self, bb_root: Path, director: "DirectorProtocol" = None):
        self._bb_root = bb_root
        self._director = director
        self._running = False
        self._task = None

    async def observe_collab(self) -> dict:
        """观察协作状态"""
        if self._director:
            return await self._director.observe()
        from teage_liu.multiagent.blackboard import read_collab_messages
        messages = await read_collab_messages(self._bb_root)
        return {"message_count": len(messages), "messages": messages}

    async def detect_anomaly(self) -> list[dict]:
        """检测异常"""
        if self._director and hasattr(self._director, "detect_anomaly"):
            return await self._director.detect_anomaly()
        return []

    async def _run_loop(self) -> None:
        """主循环：观察 + 异常检测"""
        while self._running:
            try:
                # 观察协作状态
                # state = await self.observe_collab()

                # 检测异常
                anomalies = await self.detect_anomaly()
                for anomaly in anomalies:
                    if anomaly["type"] == "timeout":
                        # 注入 intervention directive
                        if self._director:
                            await self._director.inject_directive(
                                content=f"检测到超时：{anomaly.get('detail', '')}",
                                rule_type="intervention",
                                target="*",
                            )
            except Exception:
                pass
            await asyncio.sleep(10)  # 10秒观察间隔

    async def start(self) -> None:
        """启动 Director 引擎"""
        self._running = True
        self._task = asyncio.create_task(self._run_loop())

    async def stop(self) -> None:
        """停止 Director 引擎"""
        self._running = False
        if self._task:
            self._task.cancel()
            try:
                await self._task
            except asyncio.CancelledError:
                pass
            self._task = None
```

    listEl.innerHTML = '<span class="collab-empty">暂无协作消息</span>';
    return;
  }

  listEl.innerHTML = collabMessages.map(msg => renderCollabMsgHTML(msg)).join('');
  // 滚动到底部
  listEl.scrollTop = listEl.scrollHeight;
}

// ========== 单条消息 HTML ==========
function renderCollabMsgHTML(msg) {
  const iconMap = {
    announce: '📢', request: '🎯', response: '✅', relay: '💬',
    result: '📋', directive: '⚠️', status: '📊',
  };
  const icon = iconMap[msg.type] || '•';
  const fromName = msg.from === 'user' ? '用户' : msg.from === 'director' ? 'Director' : msg.from;
  const toName = msg.to === '*' ? '所有人' : msg.to;

  let meta = '';
  if (msg.type === 'relay' && msg.to) {
    meta = ` → ${escapeHtml(toName)}`;
  }

  return `
    <div class="collab-msg ${msg.type}">
      <div class="collab-msg-header">
        <span class="collab-msg-icon">${icon}</span>
        <span>${escapeHtml(fromName)}${meta}</span>
      </div>
      <div class="collab-msg-content">${escapeHtml(msg.content || '')}</div>
    </div>
  `;
}

async def read_collab_index(bb_root: Path) -> list[dict]:
    """读取 collabs/index.md"""
    index_path = bb_root / "collabs" / "index.md"
    if not index_path.exists():
        return []
    content = index_path.read_text(encoding="utf-8")
    return _parse_frontmatter_blocks(content)


async def update_collab_index(bb_root: Path, collab_id: str,
                               title: str, status: str,
                               participants: list[str]) -> None:
    """更新协作索引（upsert）"""
    writer = _get_global_writer(bb_root)
    async with writer._lock:  # 复用全局锁
        index_path = bb_root / "collabs" / "index.md"
        index_path.parent.mkdir(parents=True, exist_ok=True)
        entries = await read_collab_index(bb_root)
        # 找到已有条目并更新，或追加新条目
        found = False
        for i, e in enumerate(entries):
            if e.get("collab_id") == collab_id:
                entries[i] = {"collab_id": collab_id, "title": title,
                              "status": status, "participants": participants}
                found = True
                break
        if not found:
            entries.append({"collab_id": collab_id, "title": title,
                            "status": status, "participants": participants})
        # 重写整个 index.md（原子写入）
        content = ""
        for e in entries:
            content += f"---\n{yaml.dump(e, allow_unicode=True, default_flow_style=False)}\n"
        await atomic_write(index_path, content)
