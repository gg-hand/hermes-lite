> 步骤用复选框(`- [x]`)语法跟踪进度。

**目标:** 接入第一个实验性扩展 = 同语言观测枝干 `audit`(声明 `observe` 能力,只读),经 `after_step` / `after` / `on_error` 钩子把对话轮摘要、对话完成、错误记录经 `host_port.storage_write` 落盘到宿主 SQLite 存储——完整走一遍"实现 → 注册 → 配置 → 消息通道 → 落盘"的接入链路,验证核心扩展机制可用。

**架构:** 观测枝干(observe)只读快照、不干预对话。钩子内数据写入走 `host_port`(InProcessHostPort)消息通道(`storage_write`,kind 带 `audit.` 前缀),由 TransportBus → StorageWriter 异步单写者落 SQLite。同语言扩展,零 `core/` 改动(接入 = 零改动铁律)。

**技术栈:** Python 3.11 async 全链路;core 为 `teage_liu2/core/`(11 钩子 + Snapshot+Action + TransportBus/InProcessHostPort + StorageWriter);存储 `SQLiteStorageProvider`。

## 全局约束

1. **接入零改动铁律**:新枝干**不得**改动 `teage_liu2/core/` 任何文件(唯一可改点 = `server/app.py` composition root 注册工厂 + 配置声明);
2. **observe 只读约束**(§15-A4③):声明 `observe` 的扩展,钩子返回 action 一律被忽略+记录——所以本枝干所有钩子一律返回 `[]`,数据写入直接 `await host_port.storage_write`;
3. **kind 前缀隔离**(§15-A3):`storage_write` 的 kind 必须以 `audit.` 开头(扩展名 + 点),跨前缀宿主拒绝;
4. **钩子异常隔离**:钩子内 await 不得向上抛(单扩展故障不影响对话)——所有存储写 try/except + `logger.warning`;
5. **配置文件改动禁止**:`config.yaml` 是常驻配置,任务 2 改动前**须经用户同意**;
6. **git 写操作须经用户批准**:本计划不含 git 提交步骤,如需提交单独征求同意;
7. **仓库纯净**:验证脚本一律放 `d:/tmp/`,用完即删,不落仓库、不 git add;
8. `extension_name` = `audit`(匹配 `^[a-z0-9_]+$`,禁点);Python 解释器用 `D:\soft\Python311\python.exe`(`python`/`py` 为 Store stub 不可用)。

- [x] **步骤 1: 创建 `teage_liu2/branches/audit.py`**

```python
"""audit 观测枝干:第一个实验性扩展(M2 轻量观测,验证接入链路)。

声明 ``observe`` 能力(只读):钩子一律返回 [] 不干预对话,数据经
``host_port.storage_write`` 消息通道落宿主存储(kind 带 ``audit.`` 前缀)。

记录三类事件:
- ``audit.steps``:每轮 LLM step 摘要(after_step,L2 StepSummary)
- ``audit.conversations``:对话完成摘要(after,终态;含终止原因)
- ``audit.errors``:对话失败记录(on_error,终态)

实验发现(2026-08-21):同语言 observe 扩展的 L3 观测投递未接线
(supervisor 仅对 stdio 异语言扩展做 l3_sink.subscribe)——本枝干刻意走
L2 通道(钩子摘要),不依赖 L3;缺口另记。
"""

from __future__ import annotations

import logging
from typing import Any, Dict, List, Optional

from teage_liu2.core.actions import Action
from teage_liu2.core.hooks import Branch
from teage_liu2.core.types import AfterResponse, Snapshot, StepSummary

logger = logging.getLogger(__name__)


class AuditBranch(Branch):
    """对话审计观测枝干:只读快照,经 host_port 落盘审计记录。"""

    name = "audit"
    #: observe 只读(§15-A4③):L3 订阅声明 + 钩子返回 action 一律忽略
    capabilities: List[str] = ["observe"]

    def __init__(self, config: dict) -> None:
        self._config: Dict[str, Any] = dict(config or {})
        self.record_steps: bool = bool(self._config.get("record_steps", True))

    async def setup(self, config: dict, host: Any) -> None:
        """枝干配置自校验(F2):配置开了却坏了 → 启动失败。"""
        if not isinstance(self.record_steps, bool):
            raise ValueError("audit.record_steps 必须是布尔值")

    # ------------------------------------------------------------------
    # 内部:经 host_port 消息通道落盘(observe 只读,不返回 action)
    # ------------------------------------------------------------------
    async def _write(self, kind: str, doc: Dict[str, Any]) -> None:
        """best-effort 落盘:失败仅告警,绝不向上抛(单扩展故障不影响对话)。"""
        if self.host_port is None:
            logger.warning("audit 未注入 host_port,跳过 %s 落盘", kind)
            return
        try:
            # kind 必须带 "audit." 前缀(§15-A3 跨前缀拒绝)
            await self.host_port.storage_write(kind, [doc])
        except Exception as e:
            logger.warning("audit 落盘失败(kind=%s): %s", kind, e)

    # ------------------------------------------------------------------
    # 钩子(全部返回 [],observe 只读约束)
    # ------------------------------------------------------------------
    async def after_step(self, snapshot: Snapshot, summary: StepSummary) -> List[Action]:
        """每轮 step 摘要(非终态钩子;observe 返回 action 会被忽略,故返回 [])。"""
        if not self.record_steps:
            return []
        await self._write("audit.steps", {
            "session_id": snapshot.session_id,
            "round": summary.round,
            "text": summary.text[:200],
            "usage": summary.usage,
            "duration": summary.duration,
        })
        return []

    async def after(self, snapshot: Snapshot, response: AfterResponse) -> List[Action]:
        """对话完成摘要(终态钩子:action 一律忽略,落盘走消息通道)。"""
        done = (response.done_event or {})
        await self._write("audit.conversations", {
            "session_id": snapshot.session_id,
            "user_input": snapshot.user_input[:200],
            "response_text": response.text[:200],
            "termination_reason": done.get("termination_reason", ""),
            "rounds": snapshot.round,
        })
        return []

    async def on_error(self, snapshot: Snapshot, error: Any) -> List[Action]:
        """对话失败记录(终态钩子)。"""
        await self._write("audit.errors", {
            "session_id": snapshot.session_id,
            "user_input": snapshot.user_input[:200],
            "error": str(error)[:500] if error is not None else "",
            "round": snapshot.round,
        })
        return []
```

```python
# d:/tmp/verify_audit_ext.py —— 验证 audit 观测枝干接入链路,用完即删
import asyncio
import json
import os
import sqlite3
import sys

# 使仓库包可导入
sys.path.insert(0, r"d:/dev/teage-liu/teage-liu")
os.environ.setdefault("PYTHONPATH", r"d:/dev/teage-liu/teage-liu")

from teage_liu2.branches.audit import AuditBranch
from teage_liu2.core.history import SQLiteHistoryStore
from teage_liu2.core.hooks import HookChain
from teage_liu2.core.pipeline import ChatPipeline
from teage_liu2.core.storage import SQLiteStorageProvider
from teage_liu2.core.storage_writer import StorageWriter
from teage_liu2.core.transport import TransportBus

# FakeLLM 参照 tests_core/fake_llm.py 的最小形态(只实现 loop 需要的流式接口)
class FakeLLM:
    def __init__(self):
        self.calls = 0
        self.activity_timeout = 60.0
        self.stream_total_timeout = 300.0

    async def chat_main_stream(self, messages, tools=None, system=None,
                               max_tokens=None, cancel_event=None,
                               activity_timeout=None, stream_total_timeout=None):
        self.calls += 1
        blocks = [{"type": "text", "text": f"回复 {self.calls}"}]
        yield {"type": "step_start", "step": self.calls}
        yield {"type": "text", "text": f"回复 {self.calls}"}
        yield {"type": "step_end", "stop_reason": "end_turn",
               "content_blocks": blocks, "usage": {"in": 10, "out": 3}}

    def close(self):
        pass


DB = r"d:/tmp/verify_audit.db"
for p in (DB, DB + "-wal", DB + "-shm"):
    if os.path.exists(p):
        os.remove(p)


async def main():
    # 1. 数据层 + 消息枢纽(模拟 app.py 装配)
    history_store = SQLiteHistoryStore(DB)
    storage_provider = SQLiteStorageProvider(DB)
    storage_writer = StorageWriter()
    fake_llm = FakeLLM()
    bus = TransportBus(llm_client=fake_llm, storage_provider=storage_provider,
                       storage_writer=storage_writer)
    bus.register_extension("audit", ["observe"])   # 身份注册(前缀/capability 授权)
    host_port = bus.make_in_process_port("audit")

    # 2. 钩子链 + audit 枝干(模拟 registry 装配 + host_port 注入)
    hooks = HookChain()
    audit = AuditBranch({"record_steps": True})
    audit.host_port = host_port
    hooks.register(audit)

    # 3. pipeline
    pipeline = ChatPipeline(llm_client=fake_llm, history_store=history_store,
                            hooks=hooks, mode="loop", max_loops=2)

    # 4. 跑一轮对话(loop 形态,1 轮 step 后 end_turn 结束)
    events = [ev async for ev in pipeline.chat_stream("session-1", "你好,审计我")]
    print("事件:", [e["type"] for e in events])
    assert events[-1]["type"] == "done", "事件流必须以 done 收尾"

    # 5. 排空 StorageWriter 写队列(等待落盘)
    storage_writer.close()

    # 6. 查库验证审计落盘
    conn = sqlite3.connect(DB)
    for kind, table in [("audit.steps", "doc_audit_steps"),
                        ("audit.conversations", "doc_audit_conversations")]:
        rows = conn.execute(f"SELECT doc_id, doc FROM {table}").fetchall()
        print(f"{kind}: {len(rows)} 条")
        for rid, doc in rows:
            print("   ", json.loads(doc))
    errors = conn.execute("SELECT doc FROM doc_audit_errors").fetchall()
    print("audit.errors:", len(errors), "条(正常对话应为 0)")
    conn.close()

    # 7. 断言
    conn = sqlite3.connect(DB)
    n_steps = conn.execute("SELECT COUNT(*) FROM doc_audit_steps").fetchone()[0]
    n_conv = conn.execute("SELECT COUNT(*) FROM doc_audit_conversations").fetchone()[0]
    n_err = conn.execute("SELECT COUNT(*) FROM doc_audit_errors").fetchone()[0]
    conn.close()
    assert n_steps >= 1, f"expected >=1 steps, got {n_steps}"
    assert n_conv >= 1, f"expected >=1 conversation, got {n_conv}"
    assert n_err == 0, f"expected 0 errors, got {n_err}"
    print("\nOK: audit 观测枝干接入链路验证通过(step + conversation 落盘, 无 error)")

    # 8. observe 只读约束验证:钩子返回 action 应被忽略(仅日志),不推进快照
    class BadObserver(AuditBranch):
        async def after_step(self, snapshot, summary):
            from teage_liu2.core.actions import SetExtra
            return [SetExtra(key="audit.bad", value=1)]

    hooks2 = HookChain()
    bad = BadObserver({"record_steps": True})
    bad.host_port = host_port
    hooks2.register(bad)
    pipeline2 = ChatPipeline(llm_client=fake_llm, history_store=history_store,
                             hooks=hooks2, mode="loop", max_loops=1)
    evs2 = [ev async for ev in pipeline2.chat_stream("session-2", "只读测试")]
    assert evs2[-1]["type"] == "done", "observe 扩展返回 action 不应破坏对话"
    print("\nOK: observe 扩展返回 action 被忽略,对话正常完成(只读约束成立)")


if __name__ == "__main__":
    asyncio.run(main())
```
